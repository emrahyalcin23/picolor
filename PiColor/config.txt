# ============================================================
# PiColor Kullanici Yapilandirmasi
# ============================================================
# Bu dosyayi SD karta kopyalayin. Cihaz her acildiginda okur;
# dosya yoksa veya SD kart takili degilse varsayilan degerler
# gecerli olur. Degisiklikler yeniden baslatmada aktif olur.
#
# Kurallar:
#   - Yorum satirlari # ile baslar, bos satirlar gozardi edilir
#   - Format: anahtar=deger (aralik boslugu olmamali)
#   - Sifredeki = isareti sorunsuz calisir: wifi_ap_pass==sifre
# ============================================================


# ============================================================
# KABLOSUZ ILETISIM
# ============================================================

# ============================================================
# AP MODU — Cihazin kendi olusturdugu Wi-Fi agi
# ============================================================
# Cihaz, asagidaki SSID ile her zaman kendi Wi-Fi agini acar.
# Dogrudan baglanti ve yapilandirma icin kullanilir.
# AP her zaman aktiftir; STA modundan bagimsizdir.

# wifi_ap_ssid — AP aginin adi (SSID).
wifi_ap_ssid=PiColor

# wifi_ap_pass — AP aginin sifresi. WPA2; en az 8 karakter.
wifi_ap_pass=picolor123

# AP IP her zaman 192.168.42.1'dir (config.h'de sabitlenmis).
# TCP portu config.h'deki DEFAULT_TCP_PORT ile belirlenir (varsayilan: 8266).
# Bu degerler yalnizca config.h'den degistirilebilir (arduino-pico kisitlamasi).


# ============================================================
# STA MODU — Ev/ofis modemi/router'a baglanti
# ============================================================
# Cihaz, asagidaki kimlik bilgileriyle ev aganiza da baglanir.
# Baglanti basarili olursa TCP sunucusu ev agindaki IP'den de
# erisilebelir olur — akilli ev sistemleri bu IP'yi kullanir.
#
# Oncelik sirasi (yukari = yuksek oncelik):
#   1. Bu dosyadaki wifi_sta_ssid / wifi_sta_pass
#   2. EEPROM'a kaydedilmis kimlik bilgileri (WIFI_SSID= komutu)
#   3. Hicbiri tanimli degilse yalnizca AP modu aktif olur.
#
# SD karti olmayan cihazlar icin: Serial Monitor'dan
#   WIFI_SSID=<ag_adi>  ve  WIFI_PASS=<sifre>  komutlarini girin.
#   Bu bilgiler EEPROM'a kaydedilir ve reset sonrasi korunur.
#
# Not: wifi_sta_ssid bos birakilirsa bu blok gozardi edilir;
#   EEPROM'daki kayit (varsa) kullanilmaya devam eder.

# wifi_sta_ssid — Ev/ofis Wi-Fi aginin adi. Bos birakin = STA pasif.
wifi_sta_ssid=

# wifi_sta_pass — Yukaridaki agin sifresi. Acik ag icin bos birakin.
wifi_sta_pass=


# ============================================================
# CIKTI FORMAT AYARLARI
# ============================================================

# basamak — Cikti degerlerindeki ondalik basamak sayisi (0-6).
#   Dusuk deger: daha yuvarlak, okunmasi kolay cikti.
#   Yuksek deger: daha hassas, veri analizi icin uygundur.
#   Ornek: basamak=0 → "47"    basamak=3 → "47.382"
#   Varsayilan: 1
basamak=1

# logaritmik — Weber-Fechner logaritmik donusumu ac/kapat.
#
#   false (kapali) — Ham degerler 0-100 arasi dogrusal (lineer)
#   olarak ciktilanir. Fiziksel isik yogunluguyla birebir orantili.
#   Acik ortamda ve nesnel karsilastirmali olcumlerde tercih edilir.
#
#   true (acik) — Insan gozunun algiladigi sekliyle cikti uretir.
#   Goz, parlaklik farklarini oransal olarak algilar: loş ısıkta
#   kucuk farklar belirgin gorunurken, cok parlak ısıkta buyuk
#   farklar bile kucuk hissedilir. Bu donusum bunu modellemek icin
#   kullanilir.
#
#   Normal log'dan farki: standart log10(x) formulu 0'i eksi
#   sonsuz yapar ve aralik bozulur. Buradaki Weber-Fechner formulu
#   f(0)=0 ve f(100)=100 sartlarini korur; aralik her zaman
#   0-100 kalir ve log_dekad parametresiyle baskı orani ayarlanir.
#
#   Varsayilan: false
logaritmik=false

# log_dekad — Logaritmik modda dinamik aralik (1.0 - 5.0).
#   Yalnizca logaritmik=true iken etkilidir.
#
#   Aralik ne kadar geniş olursa, loş ısik degerleri o kadar
#   "yukaric cekilir" ve parlak ısık degerleri o kadar "sikistirilir":
#
#     log_dekad=2 → 100:1 baskı orani  (hafif donusum)
#     log_dekad=3 → 1000:1             (onerilen, goz modeliyle uyumlu)
#     log_dekad=4 → 10000:1            (guclu baskı, karanlık ortam)
#     log_dekad=5 → 100000:1           (cok guclu baskı, nadiren gerekli)
#
#   TCS34725 sensoru icin tipik aralik: 2-4
#   Varsayilan: 3.0
log_dekad=3.0


# ============================================================
# CALISMA MODU
# ============================================================

# mod — Renk normallesirme yontemi.
#
#   STABIL  — Sabit referans maksimuma gore normalize eder (NORM_INPUT_MAX).
#   Isik kosullari degisse de olcek degismez. Farkli zamanlardaki
#   olcumleri karsilastirmak icin idealdir.
#
#   DINAMIK — Gozlemlenen maksimuma gore otomatik olceklenir.
#   Karanlik ortamda hassasiyet artar, parlak ortamda tam skala kullanilir.
#   Mutlak karsilastirma yerine anlik renk farki tespitinde tercih edilir.
#
#   Varsayilan: STABIL
mod=STABIL

# dual_cikti — Her satira hem ham (uint16_t) hem islenmiş (0-100) degeri yaz.
#   true  → Tek satirda ham sensör ve normalize degerler birlikte gelir.
#   false → Yalnizca islenmiş (normalize) degerler ciktilanir.
#   Varsayilan: true
dual_cikti=true


# ============================================================
# LOG DOSYASI (LOGGING_ENABLED=true ise gecerli)
# ============================================================

# log_file — Kullanici komutlarinin kaydedilecegi dosya yolu.
#   SD kartta bu yola her komut satirina eklenir.
log_file=/user_log.csv
